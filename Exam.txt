package user_demo;

import org.testng.Assert;
import org.testng.annotations.BeforeMethod;
import org.testng.annotations.DataProvider;
import org.testng.annotations.Test;

public class ParamTestWithDataProvider1 {

    private PrimeNumberChecker primeNumberChecker;

    // Initialize PrimeNumberChecker before each test
    @BeforeMethod
    public void initialize() {
        primeNumberChecker = new PrimeNumberChecker();
    }

    // Data provider method for test cases
    @DataProvider(name = "test1")
    public static Object[][] primeNumbers() {
        return new Object[][] {
            {2, true},
            {6, false},
            {19, true},
            {22, false},
            {23, true}
        };
    }

    // Test method that uses the data provider
    @Test(dataProvider = "test1")
    public void testPrimeNumberChecker(Integer inputNumber, Boolean expectedResult) {
        System.out.println("Input Number: " + inputNumber + " Expected Result: " + expectedResult);
        Assert.assertEquals(primeNumberChecker.validate(inputNumber), expectedResult);
    }
}

package user_demo;

public class PrimeNumberChecker {

    public Boolean validate(final Integer primeNumber) {
        if (primeNumber <= 1) {
            return false;
        }
        for (int i = 2; i <= Math.sqrt(primeNumber); i++) {
            if (primeNumber % i == 0) {
                return false;
            }
        }
        return true;
    }
}
